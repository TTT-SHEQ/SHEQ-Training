// กำหนดช่วงเวลา: 1 พ.ย. 2569 - 31 ม.ค. 2570 (ค.ศ. 2026 - 2027) const START_DATE = new Date("2026-11-01T00:00:00"); const END_DATE = new Date("2027-01-31T23:59:59"); let allAvailableSessions = []; // เก็บทุกรอบที่เป็น อังคาร, พุธ, ศุกร์ ในช่วงเวลา let visibleCount = 6; while (current <= END_DATE) { const day = current.getDay(); if (day === 2 || day === 3 || day === 5) { const yyyy = current.getFullYear(); const mm = String(current.getMonth() + 1).padStart(2, '0'); const dd = String(current.getDate()).padStart(2, '0'); const dateStr = `${yyyy}-${mm}-${dd}`; const thaiDate = current.toLocaleDateString('th-TH', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' }); dates.push({ dateStr, thaiDate, dayOfWeek: day }); } current.setDate(current.getDate() + 1); } return dates; } // ดึงจำนวนยอดจองล่าสุดจาก Google Sheets (doGet) async function fetchBookings() { const statusText = document.getElementById("seatStatusText"); const result = await response.json(); if (result && result.counts) { bookedCounts = result.counts; } if (statusText) statusText.innerText = "ข้อมูลที่นั่งเป็นปัจจุบัน"; } catch (err) { console.warn("ใช้ยอดเริ่มต้นเนื่องจากดึงข้อมูลไม่สำเร็จ:", err); if (statusText) statusText.innerText = "แสดงผลรอบเปิดรับจอง"; } finally { if (icon) icon.classList.remove("animate-spin"); renderSessions(); } } // เรนเดอร์การ์ดแสดงผลรอบวันอบรม function renderSessions() { const container = document.getElementById("sessionsContainer"); const remaining = Math.max(0, MAX_SEATS_PER_DAY - booked); const isFull = remaining <= 0; const percent = Math.min(100, (booked / MAX_SEATS_PER_DAY) * 100); const card = document.createElement("div"); card.className = `p-4 rounded-xl border transition-all cursor-pointer relative ${ isFull ? 'bg-slate-100 border-slate-300 opacity-60 cursor-not-allowed' : 'bg-white border-slate-200 hover:border-amber-500 hover:shadow-md' }`; card.id = `card-${session.dateStr}`; card.innerHTML = `
${session.dayOfWeek === 2 ? 'วันอังคาร' : session.dayOfWeek === 3 ? 'วันพุธ' : 'วันศุกร์'}

${session.thaiDate}

${isFull ? 'เต็มแล้ว' : `ว่าง ${remaining} ที่`}
14:00 - 17:00 น.
ลงทะเบียนแล้ว ${booked} คน จำกัด ${MAX_SEATS_PER_DAY} คน
`; if (!isFull) { card.onclick = () => selectSession(session); } container.appendChild(card); }); // จัดการปุ่ม "โหลดรอบวันอบรมเพิ่มเติม" let loadMoreBtn = document.getElementById("btnLoadMore"); if (!loadMoreBtn) { const btnWrapper = document.createElement("div"); btnWrapper.className = "col-span-full flex justify-center pt-3"; btnWrapper.innerHTML = ` `; container.parentNode.appendChild(btnWrapper); loadMoreBtn = document.getElementById("btnLoadMore"); } // ซ่อนปุ่มเมื่อแสดงครบทุกวันจนถึง 31 ม.ค. 70 แล้ว if (visibleCount >= allAvailableSessions.length) { if (loadMoreBtn && loadMoreBtn.parentElement) { loadMoreBtn.parentElement.style.display = "none"; } } else { if (loadMoreBtn && loadMoreBtn.parentElement) { loadMoreBtn.parentElement.style.display = "flex"; } } lucide.createIcons(); } // ฟังก์ชันกดเพิ่มรอบวันทีละ 6 วัน function loadMoreSessions() { visibleCount += 6; renderSessions(); } // ฟังก์ชันเลือกรอบวันอบรม function selectSession(session) { document.getElementById("selectedSessionDate").value = session.dateStr; const badge = document.getElementById("selectedDateBadge"); badge.classList.remove("hidden"); badge.innerText = `รอบที่เลือก: ${session.thaiDate}`; allAvailableSessions.forEach(s => { const c = document.getElementById(`card-${s.dateStr}`); if (c) { if (s.dateStr === session.dateStr) { c.classList.add("ring-2", "ring-amber-500", "bg-amber-50/40"); } else { c.classList.remove("ring-2", "ring-amber-500", "bg-amber-50/40"); } } }); document.getElementById("formSection").scrollIntoView({ behavior: 'smooth' }); } // แปลงไฟล์เป็น Base64 function fileToBase64(file) { return new Promise((resolve) => { if (!file) { resolve(null); const selectedDate = document.getElementById("selectedSessionDate").value; if (!selectedDate) { alert("กรุณาคลิกเลือกรอบวันที่ต้องการเข้าอบรมในขั้นตอนที่ 1 ก่อนครับ"); window.scrollTo({ top: 0, behavior: 'smooth' }); return; } const submitBtn = document.getElementById("submitBtn"); const btnText = document.getElementById("btnText"); submitBtn.disabled = true; btnText.innerText = "กำลังแปลงไฟล์และบันทึกลง Google Sheets..."; } catch (err) { console.error("Booking error:", err); alert("เกิดข้อผิดพลาดในการเชื่อมต่อ กรุณาลองใหม่อีกครั้ง"); submitBtn.disabled = false; btnText.innerText = "ยืนยันการจองอบรมและส่งเอกสาร"; } } function refreshSeats() { fetchBookings(); } function resetPage() { window.location.reload(); } // เริ่มต้นระบบเมื่อเปิดหน้าเว็บ window.addEventListener("DOMContentLoaded", () => { lucide.createIcons(); allAvailableSessions = generateFixedRangeSessions();
